-- Categorize orders based on sales:
-- Sales >= 4000 → High
-- Sales >= 2000 → Medium
-- Otherwise → Low
SELECT order_id, sales , IF(sales >= 4000, "High", IF(sales >= 2000, "Medium", "Low"))
AS Order_Category FROM dmart_data;

-- Categorize profit:
-- Profit >= 800 → High Profit
-- Profit >= 400 → Medium Profit
-- Otherwise → Low Profit
SELECT order_id, profit, IF(profit >= 800, "High Profit", IF(profit >= 400, "Medium Profit", "Low Profit"))
AS Profit_Category FROM dmart_data;

-- Categorize discount:
-- Discount >= 30% → High Discount
-- Discount >= 15% → Medium Discount
-- Otherwise → Low Discount
SELECT order_id, discount, IF(discount >= 30/100, "High Discount",
IF(discount >= 15/100, "Medium Discount", "Low Discount"))
AS Discount_Category FROM dmart_data;

-- Categorize customers:
-- Member → Premium Customer
-- Regular → Regular Customer
-- New → New Customer
SELECT order_id, customer_type,
CASE
WHEN customer_type = "Member" THEN "Premium Customer"
WHEN customer_type = "Regular" THEN "Regular Customer"
ELSE "New Customer"
END AS Customer_Status FROM dmart_data;

-- Categorize orders based on quantity:
-- Quantity >= 6 → Bulk Order
-- Quantity >= 3 → Medium Order
-- Otherwise → Small Order
SELECT order_id, quantity, 
CASE
WHEN quantity >= 6 THEN "Bulk Order"
WHEN quantity >= 3 THEN "Medium Order"
ELSE "Small Order"
END AS Quantity_Category FROM dmart_data;
